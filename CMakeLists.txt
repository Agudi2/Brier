cmake_minimum_required(VERSION 3.25)
project(Brier VERSION 0.1.0 LANGUAGES CXX)

if(NOT CMAKE_SYSTEM_NAME MATCHES "^(Linux|Darwin)$")
  message(FATAL_ERROR "Brier supports Linux and macOS only")
endif()
if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
  if(CMAKE_CXX_COMPILER_VERSION VERSION_LESS 13)
    message(FATAL_ERROR "GCC 13 or newer is required")
  endif()
elseif(CMAKE_CXX_COMPILER_ID MATCHES "Clang")
  if(CMAKE_CXX_COMPILER_VERSION VERSION_LESS 17)
    message(FATAL_ERROR "Clang 17 or newer is required")
  endif()
else()
  message(FATAL_ERROR "Use GCC or Clang")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(BRIER_NATIVE "Tune for the build machine" OFF)
set(BRIER_SANITIZER "none" CACHE STRING "none, address, or thread")
set_property(CACHE BRIER_SANITIZER PROPERTY STRINGS none address thread)
if(NOT BRIER_SANITIZER MATCHES "^(none|address|thread)$")
  message(FATAL_ERROR "BRIER_SANITIZER must be none, address, or thread")
endif()
include(CheckIPOSupported)
if(CMAKE_BUILD_TYPE STREQUAL "Release")
  check_ipo_supported(RESULT ipo_supported OUTPUT ipo_error)
  if(NOT ipo_supported)
    message(FATAL_ERROR "Release requires IPO/LTO: ${ipo_error}")
  endif()
  set(CMAKE_INTERPROCEDURAL_OPTIMIZATION_RELEASE ON)
endif()
add_library(brier_options INTERFACE)
target_include_directories(brier_options INTERFACE engine/include)
target_compile_options(brier_options INTERFACE -Wall -Wextra -Wpedantic -Werror
  $<$<CONFIG:Release>:-O3>)
if(BRIER_NATIVE)
  target_compile_options(brier_options INTERFACE -march=native)
endif()
if(BRIER_SANITIZER STREQUAL "address")
  target_compile_options(brier_options INTERFACE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(brier_options INTERFACE -fsanitize=address,undefined)
elseif(BRIER_SANITIZER STREQUAL "thread")
  target_compile_options(brier_options INTERFACE -fsanitize=thread -fno-omit-frame-pointer)
  target_link_options(brier_options INTERFACE -fsanitize=thread)
endif()

include(FetchContent)
FetchContent_Declare(simdjson GIT_REPOSITORY https://github.com/simdjson/simdjson.git
  GIT_TAG v3.12.2 GIT_SHALLOW TRUE)
FetchContent_Declare(googletest GIT_REPOSITORY https://github.com/google/googletest.git
  GIT_TAG v1.16.0 GIT_SHALLOW TRUE)
FetchContent_Declare(benchmark GIT_REPOSITORY https://github.com/google/benchmark.git
  GIT_TAG v1.9.1 GIT_SHALLOW TRUE)
set(BENCHMARK_ENABLE_TESTING OFF CACHE BOOL "" FORCE)
set(BENCHMARK_ENABLE_INSTALL OFF CACHE BOOL "" FORCE)
FetchContent_MakeAvailable(simdjson googletest benchmark)

find_package(Threads REQUIRED)
add_library(brier_engine STATIC engine/src/book.cpp engine/src/parse.cpp)
target_link_libraries(brier_engine PUBLIC brier_options PRIVATE simdjson Threads::Threads)

include(CTest)
if(BUILD_TESTING)
  add_executable(brier_smoke engine/tests/test_smoke.cpp)
  target_link_libraries(brier_smoke PRIVATE brier_engine GTest::gtest_main)
  include(GoogleTest)
  gtest_discover_tests(brier_smoke)
endif()
